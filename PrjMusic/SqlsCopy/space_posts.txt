-- ========== 空间说说表 ==========
CREATE TABLE space_posts (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT NOT NULL,
    content TEXT,	--内容
    images TEXT[] DEFAULT '{}',	-- 图片url
    post_type VARCHAR(20) NOT NULL DEFAULT 'original',	-- 转发/自发
    source_id BIGINT,	-- 转发源id
    source_type VARCHAR(20),		-- 转发源类型
    source_user_id BIGINT,	-- 转发源所属用户id
    extra JSONB DEFAULT '{}',		-- 存储转发信息和地址，防止原转发源被删除
    private BOOLEAN DEFAULT false,	-- 是否公开
    like_ids INT[] DEFAULT '{}',		-- 点过赞的用户
    comment_count INT DEFAULT 0,	-- 评论数
    forward_count INT DEFAULT 0,	-- 转发数
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted BOOLEAN DEFAULT FALSE		-- 逻辑删除
);

-- ========== 索引 ==========
-- 1. 查某人个人空间（最常用，按时间倒序）
CREATE INDEX idx_post_user_time 
ON space_posts(user_id, create_time DESC) 
WHERE deleted = FALSE;

-- 2. 查某条说说的所有转发（点进"转发列表"时用）
CREATE INDEX idx_post_source 
ON space_posts(source_id, create_time DESC) 
WHERE post_type = 'forward' AND deleted = FALSE;

-- 自动更新 update_time
CREATE TRIGGER update_space_posts_modtime
    BEFORE UPDATE ON space_posts
    FOR EACH ROW
    EXECUTE FUNCTION update_modified_column();
