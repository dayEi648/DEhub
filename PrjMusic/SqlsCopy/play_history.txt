-- 播放记录表

CREATE TABLE IF NOT EXISTS play_history (
    id          BIGSERIAL PRIMARY KEY,
    user_id     INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    song_id     INT NOT NULL REFERENCES musics(id) ON DELETE CASCADE,
    played_at   TIMESTAMP DEFAULT NOW()
);


-- 索引
CREATE INDEX idx_play_history_user_time ON play_history(user_id, played_at DESC);


-- 触发器函数
CREATE OR REPLACE FUNCTION handle_play_record()
RETURNS TRIGGER AS $$
DECLARE
    v_count INT;
BEGIN
    -- 删除该用户之前对这首歌的记录
    DELETE FROM play_history 
    WHERE user_id = NEW.user_id 
      AND song_id = NEW.song_id;

    -- 检查当前记录数
    SELECT COUNT(*) INTO v_count 
    FROM play_history 
    WHERE user_id = NEW.user_id;

    -- 如果已满50条，删除最旧的一条
    IF v_count >= 50 THEN
        DELETE FROM play_history
        WHERE ctid = (
            SELECT ctid FROM play_history
            WHERE user_id = NEW.user_id
            ORDER BY played_at ASC
            LIMIT 1
        );
    END IF;

    -- 允许插入新记录
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


-- 触发器
CREATE TRIGGER trg_handle_play_record
    BEFORE INSERT ON play_history
    FOR EACH ROW
    EXECUTE FUNCTION handle_play_record();
