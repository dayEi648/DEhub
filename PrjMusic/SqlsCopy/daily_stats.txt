-- ========== 每日热度增量 ==========
CREATE TABLE daily_stats (
    id BIGSERIAL PRIMARY KEY,
    target_type VARCHAR(20) NOT NULL,	--歌曲/歌单/专辑
    target_id BIGINT NOT NULL,
    stat_date DATE NOT NULL,
    daily_play_count INT DEFAULT 0,    -- 当日新增播放
    daily_collect_count INT DEFAULT 0,  -- 当日新增收藏（可选）
    daily_comment_count INT DEFAULT 0,  -- 当日新增评论（可选/专辑表不计算）
    UNIQUE(target_type, target_id, stat_date)
);

-- 聚合
CREATE INDEX idx_daily_target_date ON daily_stats(target_type, target_id, stat_date);
CREATE INDEX idx_daily_date ON daily_stats(stat_date);
