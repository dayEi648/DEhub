-- 音乐表
CREATE TABLE musics (
    id SERIAL PRIMARY KEY,
    music_name VARCHAR(32) NOT NULL,
    author_ids INT[] DEFAULT '{}' NOT NULL,
    album_id INT NULL,
    vip BOOLEAN DEFAULT FALSE NOT NULL,
    source VARCHAR(32) NULL,
    emo_tags TEXT[] DEFAULT '{}',
    interest_tags TEXT[] DEFAULT '{}',
    style VARCHAR(32) NULL,
    languages TEXT[] DEFAULT '{}',
    instruments TEXT[] DEFAULT '{}',
    collect_count INT DEFAULT 0 NOT NULL,
    hot SMALLINT DEFAULT 0 NOT NULL CHECK (hot >= 0 AND hot <= 1000),
    comment_count INT DEFAULT 0 NOT NULL,
    play_count INTEGER DEFAULT 0 NOT NULL,
    is_recommended BOOLEAN DEFAULT FALSE NOT NULL,
    is_deleted BOOLEAN DEFAULT FALSE NOT NULL,
    release_date DATE,
    file_url VARCHAR(200),
    lyrics_url VARCHAR(200),
    image1_url VARCHAR(200),
    image2_url VARCHAR(200),
    image3_url VARCHAR(200),
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL
);

-- 自动更新update_time
CREATE TRIGGER update_musics_modtime
    BEFORE UPDATE ON musics
    FOR EACH ROW
    EXECUTE FUNCTION update_modified_column();


-- 1. 等值查询索引
CREATE INDEX idx_musics_album_id ON musics (album_id);
CREATE INDEX idx_musics_style ON musics (style);

-- 2. 数组交集查询索引
CREATE INDEX idx_musics_author_ids ON musics USING gin (author_ids);

-- 3. 标签精确匹配索引
-- 注意：当前代码使用 unnest+ILIKE 模糊查询，无法利用以下索引
-- 如需利用索引，需将查询改为：emo_tags @> ARRAY['xxx']（精确匹配）
CREATE INDEX idx_musics_emo_tags ON musics USING gin (emo_tags);
CREATE INDEX idx_musics_interest_tags ON musics USING gin (interest_tags);
CREATE INDEX idx_musics_languages ON musics USING gin (languages);
CREATE INDEX idx_musics_instruments ON musics USING gin (instruments);

-- 4. 音乐名称模糊查询索引（trgm 支持 ILIKE '%xxx%'）
CREATE INDEX idx_musics_music_name_trgm ON musics USING gin (music_name gin_trgm_ops);

-- 5. 部分索引（高选择性常用查询）
-- 推荐且未删除（首页展示常用）
CREATE INDEX idx_musics_rec_active ON musics (hot DESC)
WHERE is_recommended = TRUE AND is_deleted = FALSE;
-- VIP专属音乐查询
CREATE INDEX idx_musics_vip_active ON musics (hot DESC)
WHERE vip = TRUE AND is_deleted = FALSE;

-- 排序索引
CREATE INDEX idx_musics_collect_count_asc ON musics (collect_count ASC);
CREATE INDEX idx_musics_collect_count_desc ON musics (collect_count DESC);
CREATE INDEX idx_musics_hot_asc ON musics (hot ASC);
CREATE INDEX idx_musics_hot_desc ON musics (hot DESC);
CREATE INDEX idx_musics_comment_count_asc ON musics (comment_count ASC);
CREATE INDEX idx_musics_comment_count_desc ON musics (comment_count DESC);
CREATE INDEX idx_musics_play_count_asc ON musics (play_count ASC);
CREATE INDEX idx_musics_play_count_desc ON musics (play_count DESC);
CREATE INDEX idx_musics_release_date_asc ON musics (release_date ASC);
CREATE INDEX idx_musics_release_date_desc ON musics (release_date DESC);
