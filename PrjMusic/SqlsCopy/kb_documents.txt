-- ========== 向量知识库文档表 ==========
-- 【注意】此文件为冗余备份，echovector 库的 kb_documents 表由 echoai/sqls/init_ai_tables.sql 统一创建。
-- init-db.sh 脚本不会执行本文件，修改表结构请同步更新 init_ai_tables.sql。
CREATE TABLE IF NOT EXISTS kb_documents (
    id SERIAL PRIMARY KEY,
    doc_id VARCHAR(64) NOT NULL UNIQUE,    -- 由 AI 服务生成 UUID，确保唯一
    source_type VARCHAR(32) NOT NULL,      -- 来源：manual_upload
    title VARCHAR(255),
    content TEXT NOT NULL,
    metadata JSONB DEFAULT '{}',           -- 附加元数据（原始文件名、上传者、解析方式等）
    chunk_index INTEGER DEFAULT 0,         -- 分块序号
    total_chunks INTEGER DEFAULT 1,
    embedding vector(1024),                -- text-embedding-v4，维度 1024
    is_deleted BOOLEAN DEFAULT FALSE,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_kb_docs_source ON kb_documents(source_type, is_deleted);
CREATE INDEX idx_kb_docs_doc_id ON kb_documents(doc_id);
CREATE INDEX idx_kb_docs_embedding ON kb_documents USING ivfflat (embedding vector_cosine_ops)
    WITH (lists = 100);

CREATE OR REPLACE FUNCTION update_modified_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.update_time = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ language 'plpgsql';

-- 自动更新 update_time
CREATE TRIGGER update_kb_docs_modtime
    BEFORE UPDATE ON kb_documents
    FOR EACH ROW EXECUTE FUNCTION update_modified_column();
