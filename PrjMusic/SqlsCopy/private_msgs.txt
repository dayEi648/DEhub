-- 私信表
CREATE TABLE private_message (
    id BIGSERIAL PRIMARY KEY,
    sender_id INT NOT NULL,
    receiver_id INT NOT NULL,
    conversation_key VARCHAR(50) NOT NULL, -- 例如 "1:2"
    content TEXT NOT NULL,
    is_read BOOLEAN DEFAULT FALSE,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted BOOLEAN DEFAULT FALSE
);

-- 索引
-- 1. 查某个会话的聊天记录
CREATE INDEX idx_pm_conversation 
ON private_message(conversation_key, create_time DESC);

-- 2. 查"谁给我发了未读消息"（消息列表小红点）
CREATE INDEX idx_pm_receiver_unread 
ON private_message(receiver_id, is_read, create_time DESC) 
WHERE is_read = FALSE;

-- 自动更新 update_time
CREATE TRIGGER update_private_message_modtime
    BEFORE UPDATE ON private_message
    FOR EACH ROW
    EXECUTE FUNCTION update_modified_column();
