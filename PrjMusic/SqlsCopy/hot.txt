-- ========== 热度表 ==========
CREATE TABLE hot (
    id SERIAL PRIMARY KEY,
    target_type VARCHAR(20) NOT NULL,  -- 类型: song / playlist / album
    target_id BIGINT NOT NULL,
    
    -- （从热度增量表聚合）
    play_count_7d INT DEFAULT 0,
    play_count_24h INT DEFAULT 0,
    collect_count_7d INT DEFAULT 0,
    collect_count_24h INT DEFAULT 0,
    comment_count_7d INT DEFAULT 0,
    comment_count_24h INT DEFAULT 0,
    
    -- 计算结果
    hot_score INT DEFAULT 0,	    -- 当前热度值
    -- 生成列 0冷 1温 2热 3爆
    hot_level INT GENERATED ALWAYS AS (
        CASE 
            WHEN hot_score >= 850 THEN 3
            WHEN hot_score >= 600 THEN 2
            WHEN hot_score >= 250  THEN 1
            ELSE 0
        END
    ) STORED,
    prev_hot_score INT DEFAULT 0,	-- 上一批热度值

    trend VARCHAR(10) GENERATED ALWAYS AS (
        COALESCE(
            CASE 
                WHEN prev_hot_score > hot_score THEN 'down'
                WHEN prev_hot_score < hot_score THEN 'up'
            END,
            'stable'
        )
    ) STORED,

    
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    UNIQUE(target_type, target_id),
    CONSTRAINT chk_hot_hot_score CHECK (hot_score >= 0 AND hot_score <= 1000)
);

-- 索引 
-- 总榜/分类榜
CREATE INDEX idx_hot_score ON hot(hot_score DESC);
CREATE INDEX idx_hot_type_score ON hot(target_type, hot_score DESC);

-- 自动更新update_time
CREATE TRIGGER update_hot_modtime
    BEFORE UPDATE ON hot
    FOR EACH ROW
    EXECUTE FUNCTION update_modified_column();
