
-- 用户表
CREATE TABLE users (
    id SERIAL PRIMARY KEY, 
    username VARCHAR(32) NOT NULL UNIQUE, 				-- 用户名
    password VARCHAR(255) NOT NULL,  				-- 存储加密哈希值
    name VARCHAR(32) NOT NULL, 					-- 账号昵称
    gender INTEGER DEFAULT 0 CHECK (gender >= 0 AND gender <= 2), 	-- 性别:未知/男/女
    role INTEGER DEFAULT 0 CHECK (role >= 0 AND role <= 3), 		-- 权限:用户/VIP用户/管理员/超级管理员
    status INTEGER DEFAULT 0 CHECK (status >= 0 AND status <= 3), 		-- 状态:正常/禁言/限制/封号
    safety INTEGER DEFAULT 10 CHECK (safety >= 0 AND safety <= 10), 	-- 安全指数:0~10
    is_deleted BOOLEAN DEFAULT FALSE, 				-- 是否注销

    -- 经验值系统
    exp             INTEGER DEFAULT 0 
                    CONSTRAINT chk_user_exp CHECK (exp >= 0 AND exp <= 100000),
    -- 等级（根据经验值自动生成）
    level           INTEGER GENERATED ALWAYS AS (
        CASE 
            WHEN exp >= 60000 THEN 10
            WHEN exp >= 30000 THEN 9
            WHEN exp >= 15000 THEN 8
            WHEN exp >= 8000  THEN 7
            WHEN exp >= 4000  THEN 6
            WHEN exp >= 2000  THEN 5
            WHEN exp >= 1000  THEN 4
            WHEN exp >= 600   THEN 3
            WHEN exp >= 300   THEN 2
            WHEN exp >= 100   THEN 1
            ELSE 0
        END
    ) STORED,
    -- 下一级所需经验（根据经验值自动生成）
    next_level_exp  INTEGER GENERATED ALWAYS AS (
        CASE 
            WHEN exp >= 60000 THEN 0
            WHEN exp >= 30000 THEN 60000
            WHEN exp >= 15000 THEN 30000
            WHEN exp >= 8000  THEN 15000
            WHEN exp >= 4000  THEN 8000
            WHEN exp >= 2000  THEN 4000
            WHEN exp >= 1000  THEN 2000
            WHEN exp >= 600   THEN 1000
            WHEN exp >= 300   THEN 600
            WHEN exp >= 100   THEN 300
            ELSE 100
        END
    ) STORED,
    	-- 等级进度百分比（根据经验值自动生成）
    level_progress  INTEGER GENERATED ALWAYS AS (
        CASE 
            WHEN exp >= 60000 THEN 100
            WHEN exp >= 30000 THEN ((exp - 30000) * 100 / 30000)
            WHEN exp >= 15000 THEN ((exp - 15000) * 100 / 15000)
            WHEN exp >= 8000  THEN ((exp - 8000) * 100 / 7000)
            WHEN exp >= 4000  THEN ((exp - 4000) * 100 / 4000)
            WHEN exp >= 2000  THEN ((exp - 2000) * 100 / 2000)
            WHEN exp >= 1000  THEN ((exp - 1000) * 100 / 1000)
            WHEN exp >= 600   THEN ((exp - 600) * 100 / 400)
            WHEN exp >= 300   THEN ((exp - 300) * 100 / 300)
            WHEN exp >= 100   THEN ((exp - 100) * 100 / 200)
            ELSE (exp * 100 / 100)
        END
    ) STORED,

    city VARCHAR(20), 	-- 居住地简称
    birth DATE, 	-- 生日
    description TEXT, 	-- 个人简介
    professional BOOLEAN DEFAULT FALSE, 	-- 是否专业认证
    emo_tags TEXT[] NOT NULL DEFAULT '{}',  	-- 情绪标签
    interest_tags TEXT[] NOT NULL DEFAULT '{}',  	-- 兴趣标签
    fan_ids INTEGER[] NOT NULL DEFAULT '{}',  	-- 粉丝ID
    follow_ids INTEGER[] NOT NULL DEFAULT '{}',  	-- 关注ID
    song_ids INTEGER[] NOT NULL DEFAULT '{}',  	-- 发行歌曲ID
    like_count INTEGER DEFAULT 0, 		-- 受点赞量
    avatar VARCHAR(200),  			-- 头像URL
    login_time TIMESTAMP, 			-- 登录时间
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP, 	-- 最后修改时间
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP, 	-- 创建账号时间
    collect_playlist_ids INT[] DEFAULT '{}' NOT NULL,		-- 收藏的歌单id
    collect_album_ids INT[] DEFAULT '{}' NOT NULL,		-- 收藏的专辑id
    collect_music_ids INT[] DEFAULT '{}' NOT NULL 	        -- 收藏的音乐id(收藏到用户的某个歌单，这首音乐就算是被收藏了)
);


-- 自动update_time
CREATE TRIGGER trg_update_users_timestamp
    BEFORE UPDATE ON users
    FOR EACH ROW
    EXECUTE FUNCTION update_modified_column();

-- 模糊查询索引
CREATE INDEX idx_users_name_trgm ON users USING gin(name gin_trgm_ops);
CREATE INDEX idx_users_city_trgm ON users USING gin(city gin_trgm_ops);

-- 粉丝/关注列表查询
CREATE INDEX idx_users_fan_ids ON users USING gin(fan_ids);
CREATE INDEX idx_users_follow_ids ON users USING gin(follow_ids);

-- 收藏音乐 ID 数组（按 ID 包含、交集查询时可利用）
CREATE INDEX idx_users_collect_music_ids ON users USING gin(collect_music_ids);

-- 情绪/兴趣标签
CREATE INDEX idx_users_emo_tags ON users USING gin(emo_tags);
CREATE INDEX idx_users_interest_tags ON users USING gin(interest_tags);

-- 专业认证
CREATE INDEX idx_users_professional ON users(exp DESC)
WHERE professional = TRUE;

-- 未删除用户
CREATE INDEX idx_users_active ON users(create_time DESC)
WHERE is_deleted = FALSE;

-- exp 等级排序
CREATE INDEX idx_users_exp_desc ON users(exp DESC);
CREATE INDEX idx_users_safety_desc ON users(safety DESC);
CREATE INDEX idx_users_like_count_desc ON users(like_count DESC);
CREATE INDEX idx_users_login_time_desc ON users(login_time DESC NULLS LAST);
CREATE INDEX idx_users_create_time_desc ON users(create_time DESC);
-- 粉丝量排序
CREATE INDEX idx_users_fan_count_desc ON users((array_length(fan_ids, 1)) DESC NULLS LAST);
